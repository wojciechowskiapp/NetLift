## Description

Generate a gRPC service implementation class from the generated .proto file. The service class should inherit from the proto-generated base class and be ready for dependency injection in ASP.NET Core.

---

## Acceptance Criteria

- [ ] Service class inherits from proto-generated base class (e.g., `CustomerService.CustomerServiceBase`)
- [ ] All RPC methods are overridden with proper async signatures
- [ ] Service is dependency injection ready (constructor injection pattern)
- [ ] Service is registered in DI container via extension method
- [ ] Proper cancellation token handling in all methods
- [ ] ServerCallContext properly utilized for metadata/headers
- [ ] Generate stub implementations that throw NotImplementedException
- [ ] Include XML documentation comments
- [ ] Unit tests for service generation

### Generator Implementation

```csharp
public class GrpcServiceGenerator
{
    public string Generate(ProtoServiceDefinition protoService, string targetNamespace)
    {
        var sb = new StringBuilder();

        WriteUsings(sb);
        WriteNamespace(sb, targetNamespace);
        WriteClassDeclaration(sb, protoService);
        WriteConstructor(sb, protoService);
        WriteMethods(sb, protoService);
        WriteClosingBraces(sb);

        return sb.ToString();
    }

    private void WriteClassDeclaration(StringBuilder sb, ProtoServiceDefinition service)
    {
        sb.AppendLine($"/// <summary>");
        sb.AppendLine($"/// gRPC service implementation for {service.Name}.");
        sb.AppendLine($"/// </summary>");
        sb.AppendLine($"public class {service.Name}Grpc : {service.Name}.{service.Name}Base");
        sb.AppendLine("{");
    }

    private void WriteMethod(StringBuilder sb, ProtoRpcMethod method)
    {
        sb.AppendLine($"    public override async Task<{method.ResponseType}> {method.Name}(");
        sb.AppendLine($"        {method.RequestType} request,");
        sb.AppendLine($"        ServerCallContext context)");
        sb.AppendLine("    {");
        sb.AppendLine($"        _logger.LogDebug(\"{method.Name} called\");");
        sb.AppendLine();
        sb.AppendLine("        // TODO: Implement business logic call");
        sb.AppendLine("        throw new RpcException(new Status(StatusCode.Unimplemented, \"Method not implemented\"));");
        sb.AppendLine("    }");
    }
}
```
