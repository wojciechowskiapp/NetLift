## Description

Generate Protocol Buffer (.proto) files from parsed WCF service and data contracts. Handle type mappings, message definitions, service definitions, and proto3 best practices.

---

## Acceptance Criteria

- [ ] Generate proto3 syntax .proto files
- [ ] Map WCF ServiceContract to gRPC service definition
- [ ] Map WCF OperationContract to gRPC RPC methods
- [ ] Map WCF DataContract to protobuf message
- [ ] Handle primitive type conversions (int, string, decimal, DateTime)
- [ ] Handle collections (arrays to repeated fields)
- [ ] Handle nullable types (google.protobuf wrappers)
- [ ] Generate proper field numbering (sequential from 1)
- [ ] Include XML documentation as proto comments
- [ ] Support imports (google/protobuf/timestamp.proto, etc.)
- [ ] Unit tests with complex contract scenarios

[ServiceContract(Namespace = "http://legacy.company.com/2015")]
public interface ICustomerService

    CustomerDto GetCustomer(int customerId);

    int SaveCustomer(CustomerDto customer);

public class CustomerDto

    [DataMember(Order = 3)]
    public string LastName { get; set; }

    [DataMember(Order = 4)]
    public string Email { get; set; }

    [DataMember(Order = 5)]
    public DateTime? BirthDate { get; set; }

    [DataMember(Order = 6)]
    public CustomerStatus Status { get; set; }

    [DataMember(Order = 7)]
    public decimal TotalPurchases { get; set; }
}

[DataContract]
public enum CustomerStatus

    [EnumMember] Active = 0,
    [EnumMember] Inactive = 1,
    [EnumMember] Suspended = 2

### Generator Implementation

```csharp
public class ProtoFileGenerator
{
    private readonly StringBuilder _sb = new();
    private int _indent = 0;

    public string Generate(WcfServiceContract serviceContract, List<WcfDataContract> dataContracts)
    {
        _sb.Clear();
        _indent = 0;

        WriteHeader(serviceContract);
        WriteImports(serviceContract, dataContracts);
        WriteOptions(serviceContract);
        WriteLine();

        WriteServiceDefinition(serviceContract);
        WriteLine();

        WriteRequestResponseMessages(serviceContract);
        WriteLine();

        WriteDataContracts(dataContracts);

        return _sb.ToString();
    }

    private void WriteHeader(WcfServiceContract contract)
    {
        WriteLine($"// {ToSnakeCase(contract.InterfaceName)}.proto");
        WriteLine($"// Generated by NetLift from {contract.FullTypeName}");
        WriteLine($"// Original namespace: {contract.Namespace}");
        WriteLine();
        WriteLine("syntax = \"proto3\";");
        WriteLine();
    }

    private void WriteImports(WcfServiceContract service, List<WcfDataContract> contracts)
    {
        var needsTimestamp = contracts.Any(c => c.Properties.Any(p => p.Type.Contains("DateTime")));
        var needsWrappers = contracts.Any(c => c.Properties.Any(p => p.IsNullable));

        WriteLine($"package {GetPackageName(service.Namespace)};");
        WriteLine();

        if (needsTimestamp)
            WriteLine("import \"google/protobuf/timestamp.proto\";");
        if (needsWrappers)
            WriteLine("import \"google/protobuf/wrappers.proto\";");

        if (needsTimestamp || needsWrappers)
            WriteLine();
    }

    private void WriteServiceDefinition(WcfServiceContract contract)
    {
        WriteLine($"// {contract.InterfaceName} operations");
        WriteLine($"service {RemoveIPrefix(contract.InterfaceName)} {{");
        _indent++;

        foreach (var operation in contract.Operations)
        {
            WriteServiceMethod(operation);
        }

        _indent--;
        WriteLine("}");
    }

    private void WriteServiceMethod(WcfOperation operation)
    {
        if (!string.IsNullOrEmpty(operation.XmlDocumentation))
            WriteLine($"// {operation.XmlDocumentation}");

        var requestType = $"{operation.Name}Request";
        var responseType = $"{operation.Name}Response";

        WriteLine($"rpc {operation.Name} ({requestType}) returns ({responseType});");
    }

    private void WriteRequestResponseMessages(WcfServiceContract contract)
    {
        foreach (var operation in contract.Operations)
        {
            // Request message
            WriteLine($"// Request message for {operation.Name}");
            WriteLine($"message {operation.Name}Request {{");
            _indent++;

            int fieldNumber = 1;
            foreach (var param in operation.Parameters)
            {
                var protoType = MapToProtoType(param.Type, param.IsNullable);
                WriteLine($"{protoType} {ToSnakeCase(param.Name)} = {fieldNumber++};");
            }

            _indent--;
            WriteLine("}");
            WriteLine();

            // Response message
            WriteLine($"// Response message for {operation.Name}");
            WriteLine($"message {operation.Name}Response {{");
            _indent++;

            var returnProtoType = MapToProtoType(operation.ReturnType, operation.IsNullable);

            if (returnProtoType == "void")
            {
                WriteLine("// Empty response");
            }
            else if (operation.ReturnsCollection)
            {
                WriteLine($"repeated {returnProtoType} items = 1;");
            }
            else
            {
                var fieldName = ToSnakeCase(ExtractTypeName(operation.ReturnType));
                WriteLine($"{returnProtoType} {fieldName} = 1;");
            }

            _indent--;
            WriteLine("}");
            WriteLine();
        }
    }

    private void WriteDataContracts(List<WcfDataContract> contracts)
    {
        foreach (var contract in contracts.Where(c => !c.IsEnum))
        {
            WriteMessage(contract);
            WriteLine();
        }

        foreach (var contract in contracts.Where(c => c.IsEnum))
        {
            WriteEnum(contract);
            WriteLine();
        }
    }

    private void WriteMessage(WcfDataContract contract)
    {
        WriteLine($"// {contract.ClassName}");
        WriteLine($"message {contract.ClassName} {{");
        _indent++;

        foreach (var property in contract.Properties.OrderBy(p => p.Order))
        {
            var protoType = MapToProtoType(property.Type, property.IsNullable, property.IsCollection);
            WriteLine($"{protoType} {ToSnakeCase(property.Name)} = {property.Order};");
        }

        _indent--;
        WriteLine("}");
    }

    private void WriteEnum(WcfDataContract contract)
    {
        WriteLine($"// {contract.ClassName}");
        WriteLine($"enum {contract.ClassName} {{");
        _indent++;

        // Proto3 requires first enum value to be 0
        var enumName = ToScreamingSnakeCase(contract.ClassName);
        WriteLine($"{enumName}_UNSPECIFIED = 0;");

        foreach (var member in contract.EnumMembers.OrderBy(m => m.Value))
        {
            var memberName = ToScreamingSnakeCase($"{contract.ClassName}_{member.Name}");
            WriteLine($"{memberName} = {member.Value + 1};"); // Offset by 1 due to UNSPECIFIED
        }

        _indent--;
        WriteLine("}");
    }

    private string MapToProtoType(string csharpType, bool isNullable = false, bool isCollection = false)
    {
        var baseType = csharpType.TrimEnd('[', ']', '?');

        var protoType = baseType switch
        {
            "System.Int32" or "int" => "int32",
            "System.Int64" or "long" => "int64",
            "System.String" or "string" => "string",
            "System.Boolean" or "bool" => "bool",
            "System.Double" or "double" => "double",
            "System.Decimal" or "decimal" => "double", // Proto doesn't have decimal
            "System.DateTime" or "DateTime" => "google.protobuf.Timestamp",
            "System.Byte" or "byte" => "int32",
            _ => baseType // Custom type
        };

        // Handle nullable value types with wrappers
        if (isNullable && protoType != "string")
        {
            protoType = protoType switch
            {
                "int32" => "google.protobuf.Int32Value",
                "int64" => "google.protobuf.Int64Value",
                "bool" => "google.protobuf.BoolValue",
                "double" => "google.protobuf.DoubleValue",
                _ => protoType
            };
        }

        if (isCollection)
        {
            protoType = $"repeated {protoType}";
        }

        return protoType;
    }

    private void WriteLine(string line = "")
    {
        if (string.IsNullOrEmpty(line))
        {
            _sb.AppendLine();
        }
        else
        {
            _sb.AppendLine(new string(' ', _indent * 2) + line);
        }
    }

    private string ToSnakeCase(string input) =>
        Regex.Replace(input, @"(\p{Ll})(\p{Lu})", "$1_$2").ToLowerInvariant();

    private string ToScreamingSnakeCase(string input) =>
        ToSnakeCase(input).ToUpperInvariant();

    private string RemoveIPrefix(string name) =>
        name.StartsWith("I") ? name.Substring(1) : name;
}
```

### Type Mapping Reference

| C# Type | Proto3 Type | Notes |
|---------|-------------|-------|
| `int` | `int32` | 32-bit integer |
| `long` | `int64` | 64-bit integer |
| `string` | `string` | UTF-8 string |
| `bool` | `bool` | Boolean |
| `decimal` | `double` | Loss of precision |
| `DateTime` | `google.protobuf.Timestamp` | UTC recommended |
| `int?` | `google.protobuf.Int32Value` | Nullable wrapper |
| `string[]` | `repeated string` | Collection |
| `List<T>` | `repeated T` | Collection |
| Custom class | `message` | Nested message |
| `enum` | `enum` | First value must be 0 |
