## Description

Transform WCF FaultContract definitions and FaultException handling to appropriate modern error handling patterns: gRPC status codes with rich error details, or REST ProblemDetails (RFC 7807). Generate mapping code and custom exception types.

---

- [ ] Include fault detail data in error responses

namespace ModernApp.Exceptions;

/// <summary>
/// Exception thrown when a customer is not found.
/// Migrated from WCF CustomerNotFoundFault.
/// </summary>
public class CustomerNotFoundException : BusinessException
{
    public int CustomerId { get; }

    public CustomerNotFoundException(int customerId)
        : base($"Customer with ID {customerId} was not found")
    {
        CustomerId = customerId;
    }

    public CustomerNotFoundException(int customerId, string message)
        : base(message)
    {
        CustomerId = customerId;
    }
}

// Exceptions/ValidationException.cs
public class ValidationException : BusinessException
{
    public string[] Errors { get; }
    public string Field { get; }

    public ValidationException(string[] errors, string field = null)
        : base($"Validation failed: {string.Join(", ", errors)}")
    {
        Errors = errors;
        Field = field;
    }
}

// Exceptions/AuthorizationException.cs
public class AuthorizationException : BusinessException
{
    public string RequiredPermission { get; }

    public AuthorizationException(string requiredPermission)
        : base($"Authorization failed. Required permission: {requiredPermission}")
    {
        RequiredPermission = requiredPermission;
    }
}

### Output: gRPC Status Code Mapping

```csharp
// GrpcExceptionInterceptor.cs
// Generated by NetLift

using Grpc.Core;
using Grpc.Core.Interceptors;

namespace ModernApp.Grpc.Interceptors;

/// <summary>
/// Interceptor that maps business exceptions to gRPC status codes.
/// </summary>
public class GrpcExceptionInterceptor : Interceptor
{
    private readonly ILogger<GrpcExceptionInterceptor> _logger;

    public GrpcExceptionInterceptor(ILogger<GrpcExceptionInterceptor> logger)
    {
        _logger = logger;
    }

    public override async Task<TResponse> UnaryServerHandler<TRequest, TResponse>(
        TRequest request,
        ServerCallContext context,
        UnaryServerMethod<TRequest, TResponse> continuation)
    {
        try
        {
            return await continuation(request, context);
        }
        catch (Exception ex)
        {
            throw MapToRpcException(ex);
        }
    }

    private RpcException MapToRpcException(Exception exception)
    {
        var (statusCode, message, metadata) = exception switch
        {
            CustomerNotFoundException ex => (
                StatusCode.NotFound,
                ex.Message,
                CreateMetadata(("customer_id", ex.CustomerId.ToString()))
            ),

            ValidationException ex => (
                StatusCode.InvalidArgument,
                ex.Message,
                CreateMetadata(
                    ("errors", string.Join(";", ex.Errors)),
                    ("field", ex.Field ?? "")
                )
            ),

            AuthorizationException ex => (
                StatusCode.PermissionDenied,
                ex.Message,
                CreateMetadata(("required_permission", ex.RequiredPermission))
            ),

            OperationCanceledException => (
                StatusCode.Cancelled,
                "Operation was cancelled",
                new Metadata()
            ),

            _ => (
                StatusCode.Internal,
                "An internal error occurred",
                new Metadata()
            )
        };

        _logger.LogError(exception, "gRPC call failed with {StatusCode}: {Message}", statusCode, message);

        return new RpcException(new Status(statusCode, message), metadata);
    }

    private Metadata CreateMetadata(params (string Key, string Value)[] entries)
    {
        var metadata = new Metadata();
        foreach (var (key, value) in entries)
        {
            if (!string.IsNullOrEmpty(value))
                metadata.Add(key, value);
        }
        return metadata;
    }
}
```

### Output: REST ProblemDetails Mapping

```csharp
// ProblemDetailsExceptionHandler.cs
// Generated by NetLift

using Microsoft.AspNetCore.Diagnostics;
using Microsoft.AspNetCore.Mvc;

namespace ModernApp.Api.Middleware;

/// <summary>
/// Exception handler that maps business exceptions to RFC 7807 ProblemDetails.
/// </summary>
public class ProblemDetailsExceptionHandler : IExceptionHandler
{
    private readonly ILogger<ProblemDetailsExceptionHandler> _logger;

    public ProblemDetailsExceptionHandler(ILogger<ProblemDetailsExceptionHandler> logger)
    {
        _logger = logger;
    }

    public async ValueTask<bool> TryHandleAsync(
        HttpContext httpContext,
        Exception exception,
        CancellationToken cancellationToken)
    {
        var problemDetails = MapToProblemDetails(exception);

        _logger.LogError(exception, "Request failed with {StatusCode}: {Title}",
            problemDetails.Status, problemDetails.Title);

        httpContext.Response.StatusCode = problemDetails.Status ?? 500;
        await httpContext.Response.WriteAsJsonAsync(problemDetails, cancellationToken);

        return true;
    }

    private ProblemDetails MapToProblemDetails(Exception exception)
    {
        return exception switch
        {
            CustomerNotFoundException ex => new ProblemDetails
            {
                Status = StatusCodes.Status404NotFound,
                Title = "Customer Not Found",
                Detail = ex.Message,
                Type = "https://api.example.com/errors/customer-not-found",
                Extensions = { ["customerId"] = ex.CustomerId }
            },

            ValidationException ex => new ValidationProblemDetails
            {
                Status = StatusCodes.Status400BadRequest,
                Title = "Validation Failed",
                Detail = ex.Message,
                Type = "https://api.example.com/errors/validation-failed",
                Errors = ex.Field != null
                    ? new Dictionary<string, string[]> { [ex.Field] = ex.Errors }
                    : new Dictionary<string, string[]> { [""] = ex.Errors }
            },

            AuthorizationException ex => new ProblemDetails
            {
                Status = StatusCodes.Status403Forbidden,
                Title = "Authorization Failed",
                Detail = ex.Message,
                Type = "https://api.example.com/errors/authorization-failed",
                Extensions = { ["requiredPermission"] = ex.RequiredPermission }
            },

            _ => new ProblemDetails
            {
                Status = StatusCodes.Status500InternalServerError,
                Title = "Internal Server Error",
                Detail = "An unexpected error occurred",
                Type = "https://api.example.com/errors/internal-error"
            }
        };
    }
}
```
