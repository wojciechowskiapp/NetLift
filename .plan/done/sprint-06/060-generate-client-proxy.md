## Description

Generate client proxy classes for consuming the migrated services. Support both gRPC client factory pattern and typed HTTP clients for REST APIs. Provide the same developer experience as WCF-generated proxies.

---

## Acceptance Criteria

- [ ] Generate gRPC client using GrpcClientFactory pattern
- [ ] Generate typed HTTP client for REST API
- [ ] Interface-based design matching original WCF contract
- [ ] Dependency injection registration extensions
- [ ] Retry policies and resilience (Polly integration)
- [ ] Proper error handling and exception mapping
- [ ] Cancellation token support throughout
- [ ] Include client configuration options
- [ ] Unit tests for client generation

```csharp
// CustomerServiceGrpcClient.cs
// Generated by NetLift - gRPC client implementation

using Grpc.Core;
using Grpc.Net.ClientFactory;

namespace ModernApp.Clients.Grpc;

/// <summary>
/// gRPC client implementation for CustomerService.
/// </summary>
public class CustomerServiceGrpcClient : ICustomerServiceClient
{
    private readonly CustomerService.CustomerServiceClient _grpcClient;
    private readonly ILogger<CustomerServiceGrpcClient> _logger;

    public CustomerServiceGrpcClient(
        CustomerService.CustomerServiceClient grpcClient,
        ILogger<CustomerServiceGrpcClient> logger)
    {
        _grpcClient = grpcClient ?? throw new ArgumentNullException(nameof(grpcClient));
        _logger = logger ?? throw new ArgumentNullException(nameof(logger));
    }

    public async Task<CustomerDto> GetCustomerAsync(int customerId, CancellationToken cancellationToken = default)
    {
        try
        {
            var request = new GetCustomerRequest { CustomerId = customerId };
            var response = await _grpcClient.GetCustomerAsync(request, cancellationToken: cancellationToken);
            return MapFromProto(response.Customer);
        }
        catch (RpcException ex)
        {
            _logger.LogError(ex, "gRPC call GetCustomer failed with status {Status}", ex.StatusCode);
            throw MapToClientException(ex);
        }
    }

    public async Task<int> SaveCustomerAsync(CustomerDto customer, CancellationToken cancellationToken = default)
    {
        try
        {
            var request = new SaveCustomerRequest { Customer = MapToProto(customer) };
            var response = await _grpcClient.SaveCustomerAsync(request, cancellationToken: cancellationToken);
            return response.CustomerId;
        }
        catch (RpcException ex)
        {
            _logger.LogError(ex, "gRPC call SaveCustomer failed with status {Status}", ex.StatusCode);
            throw MapToClientException(ex);
        }
    }

    public async Task<IReadOnlyList<CustomerDto>> SearchCustomersAsync(SearchCriteria criteria, CancellationToken cancellationToken = default)
    {
        try
        {
            var request = new SearchCustomersRequest { Criteria = MapToProto(criteria) };
            var response = await _grpcClient.SearchCustomersAsync(request, cancellationToken: cancellationToken);
            return response.Customers.Select(MapFromProto).ToList();
        }
        catch (RpcException ex)
        {
            _logger.LogError(ex, "gRPC call SearchCustomers failed with status {Status}", ex.StatusCode);
            throw MapToClientException(ex);
        }
    }

    public async Task<bool> DeleteCustomerAsync(int customerId, CancellationToken cancellationToken = default)
    {
        try
        {
            var request = new DeleteCustomerRequest { CustomerId = customerId };
            var response = await _grpcClient.DeleteCustomerAsync(request, cancellationToken: cancellationToken);
            return response.Success;
        }
        catch (RpcException ex)
        {
            _logger.LogError(ex, "gRPC call DeleteCustomer failed with status {Status}", ex.StatusCode);
            throw MapToClientException(ex);
        }
    }

    private Exception MapToClientException(RpcException ex)
    {
        return ex.StatusCode switch
        {
            StatusCode.NotFound => new CustomerNotFoundException(
                int.Parse(ex.Trailers.GetValue("customer_id") ?? "0")),
            StatusCode.InvalidArgument => new ValidationException(
                ex.Trailers.GetValue("errors")?.Split(';') ?? Array.Empty<string>()),
            StatusCode.PermissionDenied => new AuthorizationException(
                ex.Trailers.GetValue("required_permission") ?? "unknown"),
            _ => new ServiceException(ex.Status.Detail, ex)
        };
    }

    // Proto mapping methods
    private CustomerDto MapFromProto(CustomerProto proto) { /* ... */ }
    private CustomerProto MapToProto(CustomerDto dto) { /* ... */ }
}
```

### Output: REST HTTP Client Proxy

```csharp
// CustomerServiceHttpClient.cs
// Generated by NetLift - Typed HTTP client implementation

using System.Net.Http.Json;

namespace ModernApp.Clients.Http;

/// <summary>
/// Typed HTTP client implementation for CustomerService REST API.
/// </summary>
public class CustomerServiceHttpClient : ICustomerServiceClient
{
    private readonly HttpClient _httpClient;
    private readonly ILogger<CustomerServiceHttpClient> _logger;

    public CustomerServiceHttpClient(
        HttpClient httpClient,
        ILogger<CustomerServiceHttpClient> logger)
    {
        _httpClient = httpClient ?? throw new ArgumentNullException(nameof(httpClient));
        _logger = logger ?? throw new ArgumentNullException(nameof(logger));
    }

    public async Task<CustomerDto> GetCustomerAsync(int customerId, CancellationToken cancellationToken = default)
    {
        try
        {
            var response = await _httpClient.GetAsync($"api/customer/{customerId}", cancellationToken);
            await EnsureSuccessOrThrow(response, cancellationToken);
            return await response.Content.ReadFromJsonAsync<CustomerDto>(cancellationToken: cancellationToken);
        }
        catch (HttpRequestException ex)
        {
            _logger.LogError(ex, "HTTP call GetCustomer failed");
            throw;
        }
    }

    public async Task<int> SaveCustomerAsync(CustomerDto customer, CancellationToken cancellationToken = default)
    {
        try
        {
            var response = await _httpClient.PostAsJsonAsync("api/customer", customer, cancellationToken);
            await EnsureSuccessOrThrow(response, cancellationToken);
            return await response.Content.ReadFromJsonAsync<int>(cancellationToken: cancellationToken);
        }
        catch (HttpRequestException ex)
        {
            _logger.LogError(ex, "HTTP call SaveCustomer failed");
            throw;
        }
    }

    public async Task<IReadOnlyList<CustomerDto>> SearchCustomersAsync(SearchCriteria criteria, CancellationToken cancellationToken = default)
    {
        try
        {
            var response = await _httpClient.PostAsJsonAsync("api/customer/search", criteria, cancellationToken);
            await EnsureSuccessOrThrow(response, cancellationToken);
            var result = await response.Content.ReadFromJsonAsync<List<CustomerDto>>(cancellationToken: cancellationToken);
            return result ?? new List<CustomerDto>();
        }
        catch (HttpRequestException ex)
        {
            _logger.LogError(ex, "HTTP call SearchCustomers failed");
            throw;
        }
    }

    public async Task<bool> DeleteCustomerAsync(int customerId, CancellationToken cancellationToken = default)
    {
        try
        {
            var response = await _httpClient.DeleteAsync($"api/customer/{customerId}", cancellationToken);

            if (response.StatusCode == System.Net.HttpStatusCode.NotFound)
                return false;

            await EnsureSuccessOrThrow(response, cancellationToken);
            return true;
        }
        catch (HttpRequestException ex)
        {
            _logger.LogError(ex, "HTTP call DeleteCustomer failed");
            throw;
        }
    }

    private async Task EnsureSuccessOrThrow(HttpResponseMessage response, CancellationToken cancellationToken)
    {
        if (response.IsSuccessStatusCode)
            return;

        var problemDetails = await response.Content.ReadFromJsonAsync<ProblemDetails>(cancellationToken: cancellationToken);

        throw response.StatusCode switch
        {
            System.Net.HttpStatusCode.NotFound => new CustomerNotFoundException(
                problemDetails?.Extensions.TryGetValue("customerId", out var id) == true ? (int)id : 0),
            System.Net.HttpStatusCode.BadRequest => new ValidationException(
                new[] { problemDetails?.Detail ?? "Validation failed" }),
            System.Net.HttpStatusCode.Forbidden => new AuthorizationException(
                problemDetails?.Extensions.TryGetValue("requiredPermission", out var perm) == true ? perm.ToString() : "unknown"),
            _ => new ServiceException(problemDetails?.Detail ?? response.ReasonPhrase)
        };
    }
}
```

### DI Registration Extensions

```csharp
// ClientServiceExtensions.cs
// Generated by NetLift

using Microsoft.Extensions.DependencyInjection;
using Polly;
using Polly.Extensions.Http;

namespace ModernApp.Clients;

public static class ClientServiceExtensions
{
    /// <summary>
    /// Adds CustomerService gRPC client with resilience policies.
    /// </summary>
    public static IServiceCollection AddCustomerServiceGrpcClient(
        this IServiceCollection services,
        Action<GrpcClientOptions> configureOptions)
    {
        var options = new GrpcClientOptions();
        configureOptions(options);

        services.AddGrpcClient<CustomerService.CustomerServiceClient>(o =>
        {
            o.Address = new Uri(options.Address);
        })
        .ConfigureChannel(o =>
        {
            o.MaxReceiveMessageSize = options.MaxReceiveMessageSize;
            o.MaxSendMessageSize = options.MaxSendMessageSize;
        })
        .AddPolicyHandler(GetRetryPolicy());

        services.AddScoped<ICustomerServiceClient, CustomerServiceGrpcClient>();

        return services;
    }

    /// <summary>
    /// Adds CustomerService HTTP client with resilience policies.
    /// </summary>
    public static IServiceCollection AddCustomerServiceHttpClient(
        this IServiceCollection services,
        Action<HttpClientOptions> configureOptions)
    {
        var options = new HttpClientOptions();
        configureOptions(options);

        services.AddHttpClient<ICustomerServiceClient, CustomerServiceHttpClient>(client =>
        {
            client.BaseAddress = new Uri(options.BaseAddress);
            client.Timeout = options.Timeout;
        })
        .AddPolicyHandler(GetRetryPolicy())
        .AddPolicyHandler(GetCircuitBreakerPolicy());

        return services;
    }

    private static IAsyncPolicy<HttpResponseMessage> GetRetryPolicy()
    {
        return HttpPolicyExtensions
            .HandleTransientHttpError()
            .WaitAndRetryAsync(3, retryAttempt =>
                TimeSpan.FromSeconds(Math.Pow(2, retryAttempt)));
    }

    private static IAsyncPolicy<HttpResponseMessage> GetCircuitBreakerPolicy()
    {
        return HttpPolicyExtensions
            .HandleTransientHttpError()
            .CircuitBreakerAsync(5, TimeSpan.FromSeconds(30));
    }
}

public class GrpcClientOptions
{
    public string Address { get; set; } = "https://localhost:5001";
    public int? MaxReceiveMessageSize { get; set; }
    public int? MaxSendMessageSize { get; set; }
}

public class HttpClientOptions
{
    public string BaseAddress { get; set; } = "https://localhost:5000";
    public TimeSpan Timeout { get; set; } = TimeSpan.FromSeconds(30);
}
```
